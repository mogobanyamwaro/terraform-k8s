### 4.4 Conditional Statements

```bash
cat > conditionals.sh << 'EOF'
#!/bin/bash

# If-else statement
read -p "Enter a number: " num

if [ $num -gt 10 ]; then
    echo "$num is greater than 10"
elif [ $num -eq 10 ]; then
    echo "$num equals 10"
else
    echo "$num is less than 10"
fi

# Test operators
# Numeric: -eq, -ne, -gt, -lt, -ge, -le
# String: =, !=, -z (empty), -n (non-empty)
# File: -f (file), -d (directory), -x (executable), -r (readable), -w (writable)

# Check if file exists
file="/etc/passwd"
if [ -f "$file" ]; then
    echo "$file exists"
    if [ -r "$file" ]; then
        echo "And is readable"
    fi
fi

# Check if directory exists
if [ -d "/tmp" ]; then
    echo "/tmp is a directory"
fi

# String comparison
read -p "Enter yes or no: " answer
if [ "$answer" = "yes" ] || [ "$answer" = "y" ]; then
    echo "You said yes"
elif [ "$answer" = "no" ] || [ "$answer" = "n" ]; then
    echo "You said no"
else
    echo "Invalid answer"
fi

# Case statement
case "$answer" in
    yes|y|Y|YES)
        echo "Affirmative"
        ;;
    no|n|N|NO)
        echo "Negative"
        ;;
    *)
        echo "Unknown response"
        ;;
esac

# AND/OR conditions
if [ -f /etc/nginx/nginx.conf ] && [ -r /etc/nginx/nginx.conf ]; then
    echo "Nginx config exists and is readable"
fi
EOF

chmod +x conditionals.sh
./conditionals.sh
```

### 4.5 Loops & Iteration

```bash
cat > loops.sh << 'EOF'
#!/bin/bash

# For loop with list
echo "=== For loop with list ==="
for fruit in apple banana orange; do
    echo "Fruit: $fruit"

# For loop with range
echo "=== For loop with range ==="
for i in {1..5}; do
    echo "Number: $i"

echo "=== For loop with step ==="
for i in {0..10..2}; do
    echo "Even: $i"
done

# For loop with command output
echo "=== For loop with command output ==="
for file in *.sh; do
    echo "Script file: $file"
    if [ -x "$file" ]; then
        echo "  Executable"
    fi
done

# C-style for loop
echo "=== C-style for loop ==="
for ((i=1; i<=5; i++)); do
    echo "Iteration $i"

echo "=== While loop ==="
count=1
while [ $count -le 5 ]; do
    echo "Count: $count"
    ((count++))
done

# Until loop (opposite of while)

count=1
until [ $count -gt 5 ]; do
    echo "Count: $count"
    ((count++))
done

# Read file line by line
echo "=== Reading file ==="
while IFS= read -r line; do
    echo "Line: $line"
done < /etc/passwd | head -5

# Infinite loop with break
echo "=== Break example ==="
count=1
while true; do
    echo "Loop $count"
    if [ $count -eq 5 ]; then
        break
    fi
    ((count++))
done

# Continue example
echo "=== Continue example ==="
for i in {1..10}; do
    if [ $i -eq 5 ]; then
        continue
    fi
    echo "Number: $i"
done
EOF
```

### 4.7 Error Handling & Debugging

```bash
cat > error_handling.sh << 'EOF'
#!/bin/bash

# Set strict mode
set -e          # Exit on error
set -u          # Exit on undefined variable
set -o pipefail # Exit on pipe failures

# Alternative: set -euo pipefail

# Trap errors
error_handler() {
    echo "Error occurred on line $1"
    echo "Exit code: $2"
}

trap 'error_handler ${LINENO} $?' ERR

# Try-catch simulation with function
try() {
    if "$@"; then
        return 0
    else
        echo "Command failed: $*"
        return 1
    fi
}

# Use try
try ls /tmp
try ls /nonexistent  # This will trigger error

# Check command exit status
if ls /tmp > /dev/null 2>&1; then
    echo "Success: /tmp exists"
else
    echo "Error: /tmp does not exist"
fi

# Redirect errors
ls /nonexistent 2> /tmp/error.log
echo "Error saved to /tmp/error.log"

# Logging function
log() {
    local level=$1
    local message=$2
    local timestamp=$(date +"%Y-%m-%d %H:%M:%S")

    case $level in
        INFO)  echo "$timestamp [INFO]  $message" ;;
        WARN)  echo "$timestamp [WARN]  $message" >&2 ;;
        ERROR) echo "$timestamp [ERROR] $message" >&2 ;;
        *)     echo "$timestamp [DEBUG] $message" ;;
    esac
}

log INFO "Starting script"
log WARN "Low disk space"
log ERROR "Backup failed"

# Debug mode
debug() {
    if [ "${DEBUG:-0}" = "1" ]; then
        echo "[DEBUG] $*" >&2
    fi
}

debug "This won't show"
DEBUG=1 debug "This will show"
EOF

chmod +x error_handling.sh
./error_handling.sh
```

### 4.8 Real-World Admin Scripts

```bash
cat > admin_tools.sh << 'EOF'
#!/bin/bash
# Collection of admin tools

# Backup script
backup_dir() {
    local source=$1
    local dest=$2
    local timestamp=$(date +%Y%m%d_%H%M%S)
    local backup_file="${dest}/backup_${timestamp}.tar.gz"

    if [ ! -d "$source" ]; then
        echo "Error: Source directory $source does not exist"
        return 1
    fi

    mkdir -p "$dest"
    tar -czf "$backup_file" "$source" 2>/dev/null
    echo "Backup created: $backup_file"
}

# Log rotation script
rotate_logs() {
    local logfile=$1
    local max_size_mb=${2:-100}
    local keep_count=${3:-5}

    if [ ! -f "$logfile" ]; then
        echo "Error: Log file $logfile does not exist"
        return 1
    fi

    local size=$(du -m "$logfile" | cut -f1)
    if [ $size -gt $max_size_mb ]; then
        for ((i=$keep_count; i>=1; i--)); do
            if [ -f "${logfile}.${i}" ]; then
                mv "${logfile}.${i}" "${logfile}.$((i+1))"
            fi
        done
        mv "$logfile" "${logfile}.1"
        touch "$logfile"
        echo "Rotated $logfile (size: ${size}MB)"
    fi
}

# System health check
health_check() {
    echo "=== System Health Report ==="
    echo "Date: $(date)"
    echo
    echo "Uptime: $(uptime)"
    echo
    echo "Memory Usage:"
    free -h
    echo
    echo "Disk Usage:"
    df -h | grep -v tmpfs
    echo
    echo "Top 5 CPU Processes:"
    ps aux --sort=-%cpu | head -6
    echo
    echo "Top 5 Memory Processes:"
    ps aux --sort=-%mem | head -6
    echo
    echo "Failed Services:"
    systemctl list-units --state=failed
}

# Monitor service
monitor_service() {
    local service=$1
    local threshold=${2:-3}
    local failures=0

    while true; do
        if systemctl is-active --quiet "$service"; then
            failures=0
        else
            ((failures++))
            echo "Warning: $service is down (attempt $failures)"
            if [ $failures -ge $threshold ]; then
                echo "Alert: $service has been down for $threshold checks"
                systemctl restart "$service"
                echo "Restarted $service"
                failures=0
            fi
        fi
        sleep 30
    done
}

# Clean temp files
clean_temp() {
    local days=${1:-7}
    echo "Cleaning files older than $days days"
    find /tmp -type f -atime +$days -delete 2>/dev/null
    find /var/tmp -type f -atime +$days -delete 2>/dev/null
    echo "Cleanup complete"
}

# Menu system
show_menu() {
    echo "=== Linux Admin Tools ==="
    echo "1. Backup directory"
    echo "2. Rotate logs"
    echo "3. System health check"
    echo "4. Monitor service"
    echo "5. Clean temp files"
    echo "6. Exit"
    echo -n "Choose option: "
}

while true; do
    show_menu
    read choice
    case $choice in
        1)
            read -p "Source directory: " src
            read -p "Destination directory: " dst
            backup_dir "$src" "$dst"
            ;;
        2)
            read -p "Log file path: " logfile
            rotate_logs "$logfile"
            ;;
        3)
            health_check | tee health_report_$(date +%Y%m%d).txt
            ;;
        4)
            read -p "Service name: " service
            echo "Monitoring $service (Ctrl+C to stop)"
            monitor_service "$service"
            ;;
        5)
            clean_temp
            ;;
        6)
            echo "Goodbye!"
            exit 0
            ;;
        *)
            echo "Invalid option"
            ;;
    esac
    echo
done
EOF

chmod +x admin_tools.sh
```

```bash
# Cron syntax
# * * * * * command
# │ │ │ │ │
# │ │ │ │ └─── Day of week (0-7, 0=Sunday)
# │ │ │ └───── Month (1-12)
# │ │ └─────── Day of month (1-31)
# │ └───────── Hour (0-23)
# └─────────── Minute (0-59)

# View crontab
crontab -l

# Edit crontab
crontab -e

# Common cron examples
# Run every day at 2:30 AM
30 2 * * * /home/ubuntu/backup.sh >> /var/log/backup.log 2>&1

# Run every Monday at 3 AM
0 3 * * 1 /home/ubuntu/weekly_report.sh

# Run every 5 minutes
*/5 * * * * /home/ubuntu/check_service.sh

# Run at midnight on the first day of the month
0 0 1 * * /home/ubuntu/monthly_maintenance.sh

# Run at 8:00 AM on weekdays
0 8 * * 1-5 /home/ubuntu/daily_check.sh

# Create a backup cron job
(crontab -l 2>/dev/null; echo "0 2 * * * /home/ubuntu/backup.sh") | crontab -

# System crontabs
ls /etc/cron.d/
cat /etc/crontab
ls /etc/cron.daily/
ls /etc/cron.hourly/
ls /etc/cron.weekly/
ls /etc/cron.monthly/
```

### 4.10 Systemd Timers (Modern Cron Replacement)

```bash
# Create a timer service
cat > /tmp/backup.service << 'EOF'
[Unit]
Description=Backup Service

[Service]
Type=oneshot
ExecStart=/home/ubuntu/backup.sh
User=ubuntu
EOF

# Create the timer
cat > /tmp/backup.timer << 'EOF'
[Unit]
Description=Run backup daily
Requires=backup.service

[Timer]
OnCalendar=daily
Persistent=true

[Install]
WantedBy=timers.target
EOF

# Install

sudo cp /tmp/backup.timer /etc/systemd/system/
sudo systemctl daemon-reload
sudo systemctl enable backup.timer

# View timers
systemctl list-timers
systemctl list-timers --all

# Calendar events
systemd-analyze calendar "daily"

# Other timer examples
```

```bash
- name: Configure web servers
  hosts: webservers
  become: yes
  vars:
    http_port: 80
    max_clients: 200

  tasks:
    - name: Update apt cache
      apt:

        name: nginx
        state: present

    - name: Create website directory
      file:
        path: /var/www/mysite
        state: directory
        owner: www-data
        group: www-data
        mode: '0755'

    - name: Copy index.html
      copy:
        content: |
          <html>
            <head><title>Ansible Deployed</title></head>
            <body>
              <h1>Web Server Deployed by Ansible</h1>
              <p>Hostname: {{ ansible_hostname }}</p>
            </body>
          </html>
        dest: /var/www/mysite/index.html

    - name: Configure nginx site
      template:
        src: /tmp/mysite.j2
        dest: /etc/nginx/sites-available/mysite
      notify: restart nginx

    - name: Enable site
      file:
        src: /etc/nginx/sites-available/mysite
        dest: /etc/nginx/sites-enabled/mysite
        state: link

    - name: Ensure nginx is running
      service:
        name: nginx
        state: started
        enabled: yes

  handlers:
    - name: restart nginx
      service:
        name: nginx
        state: restarted
EOF

# Create template
cat > /tmp/mysite.j2 << 'EOF'
server {
    listen {{ http_port }};
    server_name _;
    root /var/www/mysite;
    index index.html;
}
EOF

ansible-playbook -i ~/inventory.ini ~/webserver_setup.yml
```

## Module 4 Practice Assessment

### Scenario: Create a Complete Automation Suite

```bash
cat > ~/automation_suite.sh << 'EOF'
#!/bin/bash
# Complete automation system

# Configuration
BACKUP_DIR="/backup"
LOG_DIR="/var/log/automation"
ALERT_EMAIL="admin@example.com"

# Setup
setup_environment() {
    mkdir -p $BACKUP_DIR $LOG_DIR
    touch $LOG_DIR/automation.log
    exec 1> >(tee -a $LOG_DIR/automation.log)
    exec 2>&1
}

# Logging
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1"
}

# Backup databases
backup_mysql() {
    local db_name=$1
    local backup_file="$BACKUP_DIR/mysql_${db_name}_$(date +%Y%m%d).sql.gz"

    if mysqldump "$db_name" | gzip > "$backup_file"; then
        log "SUCCESS: MySQL backup of $db_name created"
        return 0
    else
        log "ERROR: MySQL backup failed for $db_name"
        return 1
    fi
}

# Rotate logs
rotate_application_logs() {
    local log_path=$1
    local max_size_mb=100

    find "$log_path" -name "*.log" -type f -size +${max_size_mb}M | while read logfile; do
        gzip "$logfile"
        mv "$logfile.gz" "$logfile.$(date +%Y%m%d).gz"
        log "Rotated: $logfile"
    done
}

# Check system health
system_health_check() {
    local issues=0

    # Check disk space
    df -h | awk '{if($5+0 > 90) print $0}' | while read line; do
        log "WARNING: High disk usage: $line"
        ((issues++))
    done

    # Check memory
    mem_usage=$(free | grep Mem | awk '{print ($3/$2) * 100.0}' | cut -d. -f1)
    if [ $mem_usage -gt 90 ]; then
        log "WARNING: High memory usage: $mem_usage%"
        ((issues++))
    fi

    # Check failed services
    systemctl list-units --state=failed --no-legend | while read line; do
        log "ERROR: Failed service: $line"
        ((issues++))
    done

    return $issues
}

# Update system
update_system() {
    log "Starting system update"
    sudo apt update && sudo apt upgrade -y
    if [ $? -eq 0 ]; then
        log "System update completed successfully"
    else
        log "ERROR: System update failed"
        return 1
    fi
}

# Main execution
main() {
    setup_environment
    log "=== Automation Suite Started ==="

    # Run health check
    system_health_check
    health_status=$?

    # Run backups
    # backup_mysql "production_db"

    # Rotate logs
    rotate_application_logs "/var/log"

    # Update if healthy
    if [ $health_status -eq 0 ]; then
        update_system
    else
        log "Skipping updates due to system issues"
    fi

    log "=== Automation Suite Completed ==="
}

# Run main function
main
EOF

chmod +x ~/automation_suite.sh
```
