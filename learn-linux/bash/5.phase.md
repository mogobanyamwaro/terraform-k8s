## 1. Trap Signals - Cleaning Up No Matter What

The `trap` command catches signals and executes commands when your script receives them. Essential for cleaning up temporary files, releasing locks, or saving state before exit.

**Basic Signal Handling:**

```bash
# Syntax: trap 'commands' SIGNAL
trap 'echo "Script interrupted"; cleanup' INT TERM
trap 'echo "Exiting normally"' EXIT
trap 'echo "Error on line $LINENO"' ERR

# Common signals to trap:
# INT   (2)  - Ctrl+C
# TERM  (15) - Termination signal (kill)
# EXIT       - Normal script exit
# ERR        - Any non-zero exit code (with set -e)
# HUP   (1)  - Hangup (terminal closed)
# QUIT  (3)  - Quit signal (Ctrl+\)
# KILL  (9)  - CANNOT be trapped (always kills)
# STOP  (17) - CANNOT be trapped (always stops)
```

**Practical Cleanup Examples:**

```bash
#!/bin/bash
# safe-backup.sh - Backup with guaranteed cleanup

# Global variables
TEMP_DIR=""
BACKUP_FILE=""
LOG_FILE="/var/log/backup.log"

# Cleanup function - called on exit or interrupt
cleanup() {
    local exit_code=$?

    echo "[$(date)] Cleanup started. Exit code: $exit_code" >> "$LOG_FILE"

    # Remove temporary directory
    if [[ -n "$TEMP_DIR" && -d "$TEMP_DIR" ]]; then
        echo "Removing temporary directory: $TEMP_DIR"
        rm -rf "$TEMP_DIR"
    fi

    # Remove incomplete backup file
    if [[ -n "$BACKUP_FILE" && -f "$BACKUP_FILE" && $exit_code -ne 0 ]]; then
        echo "Removing incomplete backup: $BACKUP_FILE"
        rm -f "$BACKUP_FILE"
    fi

    # Release lock file
    if [[ -f /tmp/backup.lock ]]; then
        rm -f /tmp/backup.lock
    fi

    echo "[$(date)] Cleanup completed" >> "$LOG_FILE"

    # Exit with original exit code
    exit $exit_code
}

# Set up traps for various signals
trap cleanup EXIT      # Normal exit
trap cleanup INT       # Ctrl+C
trap cleanup TERM      # kill command
trap cleanup HUP       # Hangup
trap 'echo "Fatal error on line $LINENO"; exit 1' ERR

# Main script
echo "Starting backup..."
TEMP_DIR=$(mktemp -d /tmp/backup.XXXXXX)
BACKUP_FILE="/backup/backup-$(date +%Y%m%d).tar.gz"

# Simulate backup work
cp -r /etc "$TEMP_DIR/"
tar -czf "$BACKUP_FILE" -C "$TEMP_DIR" .

echo "Backup completed successfully"
```

### 2. Lock Files - Preventing Concurrent Execution

Lock files prevent multiple instances of your script from running simultaneously. Critical for cron jobs and backup scripts.

**Simple Lock File:**

```bash
#!/bin/bash
# simple-lock.sh

LOCK_FILE="/var/run/myscript.lock"

# Check if lock exists
if [[ -f "$LOCK_FILE" ]]; then
    # Get PID from lock file
    PID=$(cat "$LOCK_FILE" 2>/dev/null)

    # Check if process is still running
    if kill -0 "$PID" 2>/dev/null; then
        echo "Script already running with PID $PID. Exiting."
        exit 1
    else
        echo "Stale lock found. Removing."
        rm -f "$LOCK_FILE"
    fi
fi

# Create lock with current PID
echo $$ > "$LOCK_FILE"

# Ensure lock is removed on exit
trap 'rm -f "$LOCK_FILE"' EXIT

# Main script code here
echo "Running script..."
sleep 30
echo "Done"
```

**Advanced Locking with flock (RECOMMENDED):**

```bash

    echo "Another instance is running. Exiting."
    exit 1
fi

# Lock acquired - run script
echo "Lock acquired, running..."
trap 'flock -u 200; rm -f "$LOCK_FILE"' EXIT

# Main script

echo "Done"

# Method 2: Use flock as wrapper (in crontab)
# */5 * * * * flock -n /tmp/script.lock -c /usr/local/bin/myscript.sh

# Method 3: With timeout
if ! flock -w 10 200; then
    echo "Failed to acquire lock within 10 seconds"
    exit 1
fi

# Method 4: Show lock information
flock -n 200 || { echo "Locked by PID $(cat "$LOCK_FILE")"; exit 1; }

#!/bin/bash
# production-lock.sh - Complete lock management system

readonly SCRIPT_NAME=$(basename "$0")
readonly LOCK_DIR="/var/lock"
readonly LOCK_FILE="$LOCK_DIR/${SCRIPT_NAME}.lock"
readonly MAX_LOCK_AGE=3600  # 1 hour

acquire_lock() {
    local lock_pid

    # Create lock directory if it doesn't exist
    mkdir -p "$LOCK_DIR"

    # Check for stale lock
    if [[ -f "$LOCK_FILE" ]]; then
        lock_pid=$(cat "$LOCK_FILE" 2>/dev/null)

        # Check if PID is valid and process exists
        if [[ -n "$lock_pid" ]] && kill -0 "$lock_pid" 2>/dev/null; then
            # Process exists, check how long it's been running
            local process_age=$(ps -o etime= -p "$lock_pid" 2>/dev/null | \
                awk -F: '{if(NF==2) print $1*60+$2; else print $1*3600+$2*60+$3}')

            if [[ $process_age -gt $MAX_LOCK_AGE ]]; then
                echo "WARNING: Lock held by PID $lock_pid for > ${MAX_LOCK_AGE}s. Removing stale lock."
                rm -f "$LOCK_FILE"
            else
                echo "Script already running with PID $lock_pid (age: ${process_age}s). Exiting."
                return 1
            fi
        else
            echo "Stale lock found. Removing."
            rm -f "$LOCK_FILE"
        fi
    fi

    # Create new lock file with PID and timestamp
    echo "$$|$(date +%s)" > "$LOCK_FILE"

    # Set up trap to release lock
    trap release_lock EXIT INT TERM

    return 0
}

release_lock() {
    local my_pid=$$
    local lock_pid=$(cat "$LOCK_FILE" 2>/dev/null | cut -d'|' -f1)

    # Only release if we own the lock
    if [[ "$lock_pid" == "$my_pid" ]]; then
        rm -f "$LOCK_FILE"
        echo "Lock released"
    fi
}

show_lock_info() {
    if [[ -f "$LOCK_FILE" ]]; then
        local lock_pid=$(cat "$LOCK_FILE" | cut -d'|' -f1)
        local lock_time=$(cat "$LOCK_FILE" | cut -d'|' -f2)
        local age=$(( $(date +%s) - lock_time ))

        echo "Lock held by: PID $lock_pid"
        echo "Lock age: ${age} seconds"

        if kill -0 "$lock_pid" 2>/dev/null; then
            echo "Process is running"
            ps -p "$lock_pid" -o pid,ppid,cmd,etime
        else
            echo "WARNING: Lock process not running"
        fi
    else
        echo "No lock file found"
    fi
}

# Usage
if ! acquire_lock; then
    exit 1
fi

# For debugging
show_lock_info

# Your script here
echo "Running with lock"
sleep 60
```

### 4. Error Handling Patterns - Bulletproof Scripts

**Comprehensive Error Handling Framework:**

```bash
#!/bin/bash
# bulletproof-script.sh - Production-grade error handling

set -euo pipefail  # Strict mode
IFS=$'\n\t'        # Safe field separator

# Error handling variables
ERROR_COUNT=0
WARNING_COUNT=0
declare -a ERROR_MESSAGES

# Error handling functions
error_exit() {
    local line="$1"
    local command="$2"
    local code="${3:-1}"

    echo "Error on line $line: Command '$command' exited with status $code" >&2
    echo "Stack trace:" >&2
    for ((i=0; i<${#FUNCNAME[@]}; i++)); do
        echo "  ${FUNCNAME[$i]}() called from ${BASH_SOURCE[$i]}:${BASH_LINENO[$i]}" >&2
    done

    exit "$code"
}

warn() {
    local message="$1"
    ((WARNING_COUNT++))
    echo "[WARNING] $message" >&2
    logger -t "$(basename "$0")" -p user.warning "$message"
}

error() {
    local message="$1"
    ((ERROR_COUNT++))
    ERROR_MESSAGES+=("$message")
    echo "[ERROR] $message" >&2
    logger -t "$(basename "$0")" -p user.err "$message"
}

validate_required() {
    local var_name="$1"
    local var_value="${!var_name}"

    if [[ -z "$var_value" ]]; then
        error "Required variable $var_name is not set"
        return 1
    fi
    return 0
}

validate_file() {
    local file="$1"
    local readable="${2:-true}"

    if [[ ! -e "$file" ]]; then
        error "File does not exist: $file"
        return 1
    fi

    if [[ "$readable" == "true" ]] && [[ ! -r "$file" ]]; then
        error "File is not readable: $file"
        return 1
    fi

    return 0
}

validate_directory() {
    local dir="$1"
    local writable="${2:-false}"

    if [[ ! -d "$dir" ]]; then
        error "Directory does not exist: $dir"
        return 1
    fi

    if [[ "$writable" == "true" ]] && [[ ! -w "$dir" ]]; then
        error "Directory is not writable: $dir"
        return 1
    fi

    return 0
}

retry_command() {
    local max_attempts="$1"
    local delay="$2"
    shift 2
    local command="$*"

    local attempt=1

    while [[ $attempt -le $max_attempts ]]; do
        if eval "$command"; then
            return 0
        fi

        warn "Command failed (attempt $attempt/$max_attempts): $command"

        if [[ $attempt -eq $max_attempts ]]; then
            error "Command failed after $max_attempts attempts: $command"
            return 1
        fi

        sleep "$delay"
        ((attempt++))
    done
}

check_disk_space() {
    local path="$1"
    local required_mb="$2"

    local available_mb=$(df -m "$path" | awk 'NR==2 {print $4}')

    if [[ $available_mb -lt $required_mb ]]; then
        error "Insufficient disk space on $path: ${available_mb}MB available, ${required_mb}MB required"
        return 1
    fi

    return 0
}

check_memory() {
    local required_mb="$1"

    local available_mb=$(free -m | awk 'NR==2 {print $7}')

    if [[ $available_mb -lt $required_mb ]]; then
        error "Insufficient memory: ${available_mb}MB available, ${required_mb}MB required"
        return 1
    fi

    return 0
}

timeout_command() {
    local timeout_sec="$1"
    shift
    local command="$*"

    # Run command in background
    eval "$command" &
    local cmd_pid=$!

    # Wait with timeout
    local counter=0
    while kill -0 "$cmd_pid" 2>/dev/null && [[ $counter -lt $timeout_sec ]]; do
        sleep 1
        ((counter++))
    done

    # Kill if still running
    if kill -0 "$cmd_pid" 2>/dev/null; then
        warn "Command timed out after ${timeout_sec}s, killing: $command"
        kill -9 "$cmd_pid" 2>/dev/null
        return 124  # Standard timeout exit code
    fi

    wait "$cmd_pid"
    return $?
}

# Trap errors for detailed reporting
trap 'error_exit $LINENO "$BASH_COMMAND" $?' ERR

# Example usage
main() {
    log() { echo "[$(date)] $*"; }

    log "Starting bulletproof script"

    # Validate requirements
    validate_directory "/backup" true || exit 1
    check_disk_space "/backup" 1024 || exit 1  # Need 1GB free

    # Retry network operations
    retry_command 3 5 "wget -O /tmp/test.html http://example.com" || exit 1

    # Run with timeout
    timeout_command 30 "mysqldump --all-databases > /backup/db.sql" || exit 1

    # Check results
    if [[ $ERROR_COUNT -gt 0 ]]; then
        log "Script completed with $ERROR_COUNT errors and $WARNING_COUNT warnings"
        for msg in "${ERROR_MESSAGES[@]}"; do
            log "  ERROR: $msg"
        done
        exit 1
    fi

    log "Script completed successfully"
}

main "$@"
```

#!/bin/bash
# graceful-degradation.sh

# Try optional features, continue if they fail
enable_optional_feature() {
    if feature_available; then
        log "INFO" "Enabling optional feature"
        # Enable feature
    else
        log "WARNING" "Optional feature not available, continuing without it"
    fi
}

# Fallback to alternative
get_network_config() {
    if [[ -f /etc/netplan/*.yaml ]]; then
        cat /etc/netplan/*.yaml
    elif [[ -f /etc/network/interfaces ]]; then
        cat /etc/network/interfaces
    else
        warn "No network configuration found"
        return 1
    fi
}

# Use default values when missing
SERVER="${SERVER:-localhost}"
PORT="${PORT:-8080}"
TIMEOUT="${TIMEOUT:-30}"

# Try multiple methods
get_user_home() {
    local username="$1"
    local home=""

    # Method 1: getent
    home=$(getent passwd "$username" | cut -d: -f6 2>/dev/null)

    # Method 2: /etc/passwd
    if [[ -z "$home" ]]; then
        home=$(grep "^$username:" /etc/passwd | cut -d: -f6)
    fi

    # Method 3: default
    if [[ -z "$home" ]]; then
        home="/home/$username"
        warn "Using default home directory for $username: $home"
    fi

    echo "$home"
}

# Circuit breaker pattern
circuit_breaker() {
    local service="$1"
    local failure_file="/tmp/${service}.failures"
    local max_failures=3
    local reset_minutes=5

    local failures=$(cat "$failure_file" 2>/dev/null || echo 0)
    local last_failure=$(stat -c %Y "$failure_file" 2>/dev/null || echo 0)
    local now=$(date +%s)

    # Reset counter if older than reset_minutes
    if [[ $((now - last_failure)) -gt $((reset_minutes * 60)) ]]; then
        failures=0
    fi

    if [[ $failures -ge $max_failures ]]; then
        error "Circuit breaker open for $service (failed ${failures}x)"
        return 1
    fi

    # Call the actual service
    if call_service "$service"; then
        # Success - reset counter
        rm -f "$failure_file"
        return 0
    else
        # Failure - increment counter
        echo $((failures + 1)) > "$failure_file"
        return 1
    fi
}

#!/bin/bash
# self-healing.sh - Scripts that fix themselves

# Function to check and fix common issues
self_heal() {
    local issue="$1"

    case "$issue" in
        "missing_config")
            log "WARNING" "Config missing, restoring from template"
            cp /etc/templates/app.conf /etc/app/app.conf
            ;;
        "permission_denied")
            log "WARNING" "Permission issue, fixing ownership"
            chown -R appuser:appgroup /var/lib/app
            chmod 755 /var/lib/app
            ;;
        "disk_full")
            log "WARNING" "Disk full, cleaning old logs"
            find /var/log -name "*.log" -mtime +30 -delete
            find /tmp -type f -atime +7 -delete
            ;;
        "process_zombie")
            log "WARNING" "Zombie process detected, cleaning up"
            kill -CHLD $(pgrep -P 1) 2>/dev/null
            ;;
        *)
            log "ERROR" "Unknown issue: $issue"
            return 1
            ;;
    esac

    log "INFO" "Self-healing applied for: $issue"
    return 0
}

# Run command with automatic retry and self-heal
robust_run() {
    local max_attempts=3
    local attempt=1

    while [[ $attempt -le $max_attempts ]]; do
        if "$@"; then
            return 0
        fi

        local exit_code=$?

        # Try to identify and fix the issue
        if [[ $exit_code -eq 1 ]]; then
            # Check for common issues
            if ! command -v "${1}" &>/dev/null; then
                self_heal "missing_command"
            elif [[ ! -f "config.ini" ]]; then
                self_heal "missing_config"
            elif [[ ! -w "." ]]; then
                self_heal "permission_denied"
            elif [[ $(df -h / | awk 'NR==2 {print $5}' | sed 's/%//') -gt 90 ]]; then
                self_heal "disk_full"
            fi
        fi

        ((attempt++))
        sleep $((attempt * 2))
    done

    error "Failed after $max_attempts attempts: $*"
    return 1
}

# Health check that fixes problems
health_check() {
    local services=("nginx" "mysql" "php-fpm")

    for service in "${services[@]}"; do
        if ! systemctl is-active --quiet "$service"; then
            log "WARNING" "$service is not running, attempting restart"

            if systemctl start "$service"; then
                log "INFO" "Successfully restarted $service"
            else
                log "ERROR" "Failed to restart $service"
                # Escalate to alert
                send_alert "Service $service is down and won't restart"
            fi
        fi
    done

    # Check disk and clean if needed
    local disk_usage=$(df -h / | awk 'NR==2 {print $5}' | sed 's/%//')
    if [[ $disk_usage -gt 85 ]]; then
        log "WARNING" "Disk usage at ${disk_usage}%, cleaning up"

        # Clean package cache
        apt-get clean 2>/dev/null || yum clean all 2>/dev/null

        # Remove old kernels (Debian/Ubuntu)
        apt-get autoremove --purge -y 2>/dev/null

        # Remove orphaned packages
        package-cleanup --orphans 2>/dev/null
    fi
}

START_TIME=$(date +%s)
declare -a ERROR_MESSAGES
declare -a WARNING_MESSAGES

# ============================================
# Logging Functions
# ============================================

log() {
    local level="$1"
    shift
    local message="$*"
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')

    # Always write to log file
    echo "[$timestamp] [$level] $message" >> "$LOG_FILE"

    # Output to console if interactive
    if [[ -t 1 ]]; then
        case "$level" in
            ERROR|CRITICAL|FATAL)
                echo -e "\033[31m[$level] $message\033[0m" >&2
                ;;
            WARNING)
                echo -e "\033[33m[$level] $message\033[0m" >&2
                ;;
            INFO)
                echo -e "\033[32m[$level] $message\033[0m"
                ;;
            DEBUG)
                [[ "$LOG_LEVEL" == "DEBUG" ]] && echo "[DEBUG] $message"
                ;;
        esac
    fi

    # Send to syslog
    logger -t "$SCRIPT_NAME" -p "user.${level,,}" "$message"
}

log_info() { log "INFO" "$*"; }
log_warn() {
    log "WARNING" "$*"
    ((WARNING_COUNT++))
    WARNING_MESSAGES+=("$*")
}
log_error() {
    log "ERROR" "$*"
    ((ERROR_COUNT++))
    ERROR_MESSAGES+=("$*")
}
log_debug() { [[ "$LOG_LEVEL" == "DEBUG" ]] && log "DEBUG" "$*"; }

# ============================================
# Error Handling
# ============================================

cleanup() {
    local exit_code=$?

    log_info "Starting cleanup (exit code: $exit_code)"

    # Remove temporary files
    if [[ -n "${TEMP_DIR:-}" && -d "$TEMP_DIR" ]]; then
        rm -rf "$TEMP_DIR"
        log_debug "Removed temp directory: $TEMP_DIR"
    fi

    # Release lock
    if [[ -f "$LOCK_FILE" ]] && [[ "$(cat "$LOCK_FILE" 2>/dev/null)" == "$PID" ]]; then
        rm -f "$LOCK_FILE"
        log_debug "Released lock file"
    fi

    # Log execution time
    local end_time=$(date +%s)
    local duration=$((end_time - START_TIME))
    log_info "Execution time: ${duration} seconds"

    # Log summary
    if [[ $exit_code -eq 0 ]]; then
        log_info "Script completed successfully (Warnings: $WARNING_COUNT)"
    else
        log_error "Script failed with exit code $exit_code (Errors: $ERROR_COUNT, Warnings: $WARNING_COUNT)"
        for msg in "${ERROR_MESSAGES[@]}"; do
            log_error "  - $msg"
        done
    fi

    exit $exit_code
}

error_handler() {
    local line=$1
    local command=$2
    local code=$3

    log_error "Line $line: Command '$command' failed with exit code $code"

    # Show stack trace
    log_debug "Stack trace:"
    for ((i=0; i<${#FUNCNAME[@]}; i++)); do
        log_debug "  ${FUNCNAME[$i]}() called from ${BASH_SOURCE[$i]}:${BASH_LINENO[$i]}"
    done
}

# Set up traps
trap cleanup EXIT
trap 'error_handler $LINENO "$BASH_COMMAND" $?' ERR
trap 'log_warn "Script interrupted by user"; exit 130' INT TERM

# ============================================
# Lock Management
# ============================================

acquire_lock() {
    log_debug "Attempting to acquire lock: $LOCK_FILE"

    # Check for stale lock
    if [[ -f "$LOCK_FILE" ]]; then
        local lock_pid=$(cat "$LOCK_FILE" 2>/dev/null)

        if [[ -n "$lock_pid" ]] && kill -0 "$lock_pid" 2>/dev/null; then
            # Check lock age
            local lock_age=$(($(date +%s) - $(stat -c %Y "$LOCK_FILE" 2>/dev/null || echo 0)))

            if [[ $lock_age -gt $MAX_LOCK_AGE ]]; then
                log_warn "Stale lock found (age: ${lock_age}s), removing"
                rm -f "$LOCK_FILE"
            else
                log_error "Script already running with PID $lock_pid (age: ${lock_age}s)"
                return 1
            fi
        else
            log_warn "Removing stale lock (PID $lock_pid not running)"
            rm -f "$LOCK_FILE"
        fi
    fi

    # Create lock file
    echo "$PID" > "$LOCK_FILE"
    log_debug "Lock acquired (PID: $PID)"
    return 0
}

# ============================================
# Utility Functions
# ============================================

validate_config() {
    if [[ ! -f "$CONFIG_FILE" ]]; then
        log_error "Configuration file not found: $CONFIG_FILE"
        return 1
    fi

    # Source configuration
    source "$CONFIG_FILE"
    log_info "Loaded configuration from $CONFIG_FILE"

    return 0
}

check_prerequisites() {
    local required_commands=("$@")
    local missing=()

    for cmd in "${required_commands[@]}"; do
        if ! command -v "$cmd" &>/dev/null; then
            missing+=("$cmd")
        fi
    done

    if [[ ${#missing[@]} -gt 0 ]]; then
        log_error "Missing required commands: ${missing[*]}"
        return 1
    fi

    return 0
}

retry() {
    local max_attempts="$1"
    local delay="$2"
    shift 2
    local cmd="$*"

    local attempt=1

    while [[ $attempt -le $max_attempts ]]; do
        log_debug "Attempt $attempt/$max_attempts: $cmd"

        if eval "$cmd"; then
            return 0
        fi

        if [[ $attempt -lt $max_attempts ]]; then
            log_warn "Command failed, retrying in ${delay}s (attempt $attempt/$max_attempts)"
            sleep "$delay"
        fi

        ((attempt++))
    done

    log_error "Command failed after $max_attempts attempts: $cmd"
    return 1
}

check_disk_space() {
    local path="${1:-/}"
    local required_mb="${2:-100}"

    local available_mb=$(df -m "$path" | awk 'NR==2 {print $4}')

    if [[ $available_mb -lt $required_mb ]]; then
        log_error "Insufficient disk space on $path: ${available_mb}MB available, ${required_mb}MB required"
        return 1
    fi

    log_debug "Disk space OK: ${available_mb}MB available on $path"
    return 0
}

check_memory() {
    local required_mb="${1:-256}"

    local available_mb=$(free -m | awk 'NR==2 {print $7}')

    if [[ $available_mb -lt $required_mb ]]; then
        log_error "Insufficient memory: ${available_mb}MB available, ${required_mb}MB required"
        return 1
    fi

    log_debug "Memory OK: ${available_mb}MB available"
    return 0
}

# ============================================
# Main Script Logic
# ============================================

main() {
    log_info "=== $SCRIPT_NAME starting (PID: $PID) ==="

    # Initialization
    acquire_lock || exit 1
    validate_config || exit 1

    # Check prerequisites
    check_prerequisites "tar" "gzip" "mysql" || exit 1

    # Check system resources
    check_disk_space "/" 1024 || exit 1
    check_memory 512 || exit 1

    # Create temporary directory
    TEMP_DIR=$(mktemp -d "/tmp/${SCRIPT_NAME}.XXXXXX")
    log_debug "Created temp directory: $TEMP_DIR"

    # Main logic here
    log_info "Executing main tasks..."

    # Example: Backup operation
    if ! retry 3 5 "tar -czf $TEMP_DIR/backup.tar.gz /etc 2>/dev/null"; then
        log_error "Backup failed"
        exit 1
    fi

    # Example: Database backup
    if command -v mysqldump &>/dev/null; then
        retry 2 10 "mysqldump --all-databases > $TEMP_DIR/db.sql" || log_warn "Database backup failed"
    fi

    # Success
    log_info "Main tasks completed"

    # Validation
    if [[ ! -f "$TEMP_DIR/backup.tar.gz" ]]; then
        log_error "Backup file not created"
        exit 1
    fi

    local backup_size=$(du -h "$TEMP_DIR/backup.tar.gz" | cut -f1)
    log_info "Backup size: $backup_size"

    log_info "=== $SCRIPT_NAME completed successfully ==="
}

# ============================================
# Command Line Arguments
# ============================================

usage() {
    cat << EOF
Usage: $0 [OPTIONS]

Options:
    -c FILE     Configuration file (default: $CONFIG_FILE)
    -l LEVEL    Log level (DEBUG, INFO, WARNING, ERROR)
    -f FILE     Log file (default: $LOG_FILE)
    -h          Show this help
    -v          Show version

Examples:
    $0
    $0 -c /etc/myapp/custom.conf -l DEBUG
EOF
    exit 0
}

version() {
    echo "$SCRIPT_NAME version 1.0"
    exit 0
}

# Parse arguments
while getopts "c:l:f:hv" opt; do
    case $opt in
        c) CONFIG_FILE="$OPTARG" ;;
        l) LOG_LEVEL="$OPTARG" ;;
        f) LOG_FILE="$OPTARG" ;;
        h) usage ;;
        v) version ;;
        *) usage ;;
    esac
done

# ============================================
# Execute
# ============================================

main "$@"
