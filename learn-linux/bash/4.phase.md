### 1. Arrays - Managing Lists of Data

Arrays let you store multiple values in a single variable. Essential for admin scripts dealing with lists of servers, files, or users.

**Regular (Indexed) Arrays:**

```bash
# Declaration methods

array2=([0]="first" [1]="second")     # Explicit indices

echo "${array1[1]}"                    # two (second element)
echo "${array1[-1]}"                   # four (last element)

# All elements
echo "${array1[@]}"                    # "one two three four"
echo "${array1[*]}"                    # "one two three four" (slightly different)

# Array length (number of elements)
echo "${#array1[@]}"                   # 4
echo "${#array1[*]}"                   # 4

# Length of a specific element
echo "${#array1[0]}"                   # 3 (length of "one")
```

### 3. Parameter Expansion - Text Manipulation Without External Commands

Parameter expansion is Bash's built-in string manipulation. Much faster than forking `sed` or `awk` for simple operations.

**Basic Parameter Expansion:**

```bash
# ${var:-default} - Use default value if var is unset or null
echo "${name:-John}"    # If name not set, use "John"
echo "${name-John}"     # Use default only if unset (not null)

# ${var:=default} - Assign default if unset or null
echo "${name:=John}"    # Sets name to John if not set
echo "$name"            # Now John

# ${var:?error} - Show error if var is unset or null
echo "${config_file:?ERROR: config_file not set}"

# ${var:+alternate} - Use alternate if var is set and not null
debug_mode="true"
echo "${debug_mode:+Debug mode is ON}"  # Shows message if debug_mode is set

# Length of variable
text="Hello World"
echo "${#text}"         # 11

# Real admin example
backup_dir="${1:-/backup}"
echo "Using backup directory: $backup_dir"

# Required parameter with error
server="${1:?ERROR: Server name required}"
```

**Substring Extraction:**

```bash
#!/bin/bash
# advanced-backup-system.sh - Production backup script using arrays, functions, associative arrays

set -euo pipefail  # Strict mode

# Configuration
declare -A CONFIG
CONFIG=(
    ["backup_root"]="/backup"
    ["retention_days"]="30"
    ["parallel_jobs"]="4"
    ["log_file"]="/var/log/backup.log"
)

# Color definitions for output
declare -A COLORS=(
    ["red"]='\033[0;31m'
    ["green"]='\033[0;32m'
    ["yellow"]='\033[1;33m'
    ["blue"]='\033[0;34m'
    ["nc"]='\033[0m'
)

# Sources to backup (array of directories)
declare -a SOURCES=(
    "/etc"
    "/home"
    "/var/www"
    "/opt/application/data"
)

# Database configurations (associative array)
declare -A DATABASES=(
    ["production"]="prod_db"
    ["staging"]="staging_db"
    ["reporting"]="reporting_db"
)

# Logging function with colors
log() {
    local level="$1"
    local message="$2"
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    local color="${COLORS[green]}"

    case "$level" in
        ERROR) color="${COLORS[red]}" ;;
        WARNING) color="${COLORS[yellow]}" ;;
        INFO) color="${COLORS[green]}" ;;
        DEBUG) color="${COLORS[blue]}" ;;
    esac

    echo -e "${color}[$timestamp] [$level]${COLORS[nc]} $message" | tee -a "${CONFIG[log_file]}"
}

# Function to create backup directory structure
setup_backup_dirs() {
    local backup_date=$(date +%Y%m%d_%H%M%S)
    local backup_path="${CONFIG[backup_root]}/$backup_date"

    mkdir -p "$backup_path"/{files,databases,metadata}
    echo "$backup_path"
}

# Function to backup files with parallel processing
backup_files() {
    local backup_path="$1"
    local pids=()
    local job_count=0

    log "INFO" "Starting file backup of ${#SOURCES[@]} sources"

    for source in "${SOURCES[@]}"; do
        # Skip if source doesn't exist
        if [[ ! -e "$source" ]]; then
            log "WARNING" "Source $source does not exist, skipping"
            continue
        fi

        local source_name="${source#/}"
        source_name="${source_name//\//_}"
        local backup_file="${backup_path}/files/${source_name}.tar.gz"

        # Run backup in background
        {
            log "INFO" "Backing up $source to $backup_file"
            if tar -czf "$backup_file" "$source" 2>/dev/null; then
                log "INFO" "Successfully backed up $source"
            else
                log "ERROR" "Failed to backup $source"
            fi
        } &

        pids+=($!)
        ((job_count++))

        # Limit parallel jobs
        if [[ $job_count -ge ${CONFIG[parallel_jobs]} ]]; then
            wait "${pids[@]}"
            pids=()
            job_count=0
        fi
    done

    # Wait for remaining jobs
    wait "${pids[@]}"
    log "INFO" "File backup completed"
}

    log "INFO" "Starting database backups"

    for db_name in "${!DATABASES[@]}"; do
        local db_value="${DATABASES[$db_name]}"
        local backup_file="${backup_path}/databases/${db_name}_${db_value}.sql.gz"

        log "INFO" "Backing up database $db_name ($db_value)"

        # Simulated database backup (replace with actual mysqldump)
        # mysqldump -u backup_user "$db_value" | gzip > "$backup_file"

        # For demo, create dummy backup
        echo "DUMP OF $db_value" | gzip > "$backup_file"

        if [[ -f "$backup_file" && -s "$backup_file" ]]; then
            local size=$(du -h "$backup_file" | cut -f1)
            log "INFO" "Database $db_name backed up (size: $size)"
        else
            log "ERROR" "Failed to backup database $db_name"
        fi
    done
}

# Function to create metadata
create_metadata() {
    local backup_path="$1"
    local metadata_file="${backup_path}/metadata/backup_info.txt"

    log "INFO" "Creating metadata file"

    cat << EOF > "$metadata_file"
Backup Information
==================
Date: $(date)
Hostname: $(hostname)
User: $(whoami)
Backup Path: $backup_path

System Information:
------------------
OS: $(uname -a)
Uptime: $(uptime)
Memory: $(free -h)
Disk: $(df -h)

Backup Sources:
---------------
EOF

    # Add sources
    for source in "${SOURCES[@]}"; do
        echo "- $source" >> "$metadata_file"
    done

    echo -e "\nDatabases:" >> "$metadata_file"
    for db_name in "${!DATABASES[@]}"; do
        echo "- $db_name: ${DATABASES[$db_name]}" >> "$metadata_file"
    done

    # Calculate total backup size
    local total_size=$(du -sh "$backup_path" | cut -f1)
    echo -e "\nTotal Backup Size: $total_size" >> "$metadata_file"

    log "INFO" "Metadata created at $metadata_file"
}

# Function to clean old backups
cleanup_old_backups() {
    local retention_days="${CONFIG[retention_days]}"

    log "INFO" "Cleaning backups older than $retention_days days"

    local deleted=0
    while IFS= read -r backup_dir; do
        if [[ -d "$backup_dir" ]]; then
            log "INFO" "Deleting old backup: $backup_dir"
            rm -rf "$backup_dir"
            ((deleted++))
        fi
    done < <(find "${CONFIG[backup_root]}" -maxdepth 1 -type d -mtime +"$retention_days" -name "20*")

    log "INFO" "Deleted $deleted old backups"
}

# Function to verify backup integrity
verify_backup() {
    local backup_path="$1"
    local errors=0

    log "INFO" "Verifying backup integrity"

    # Check each tar file
    for tar_file in "$backup_path"/files/*.tar.gz; do
        if [[ -f "$tar_file" ]]; then
            log "DEBUG" "Verifying $tar_file"
            if ! tar -tzf "$tar_file" >/dev/null 2>&1; then
                log "ERROR" "Corrupt backup: $tar_file"
                ((errors++))
            fi
        fi
    done

    # Check database backups
    for sql_file in "$backup_path"/databases/*.sql.gz; do
        if [[ -f "$sql_file" ]]; then
            if ! gzip -t "$sql_file" 2>/dev/null; then
                log "ERROR" "Corrupt database backup: $sql_file"
                ((errors++))
            fi
        fi
    done

    if [[ $errors -eq 0 ]]; then
        log "INFO" "Backup verification passed"
        return 0
    else
        log "ERROR" "Backup verification failed with $errors errors"
        return 1
    fi
}

# Function to send summary email
send_summary() {
    local backup_path="$1"
    local status="$2"

    local subject="Backup $status - $(hostname) - $(date +%Y%m%d)"

    mail -s "$subject" admin@example.com << EOF
Backup Summary
==============
Status: $status
Date: $(date)
Host: $(hostname)
Backup Location: $backup_path
Log File: ${CONFIG[log_file]}

Last 10 log entries:
$(tail -10 "${CONFIG[log_file]}")

Please check the log file for details.
EOF

    log "INFO" "Summary email sent"
}

# Main function
main() {
    local start_time=$(date +%s)

    log "INFO" "=== Starting backup process ==="

    # Validate configuration
    if [[ ! -d "${CONFIG[backup_root]}" ]]; then
        log "INFO" "Creating backup root directory: ${CONFIG[backup_root]}"
        mkdir -p "${CONFIG[backup_root]}"
    fi

    # Setup backup directory
    local backup_path=$(setup_backup_dirs)
    log "INFO" "Backup path: $backup_path"

    # Perform backups
    backup_files "$backup_path"
    backup_databases "$backup_path"
    create_metadata "$backup_path"

    # Verify and cleanup
    if verify_backup "$backup_path"; then
        cleanup_old_backups
        local status="SUCCESS"
    else
        local status="FAILED"
    fi

    # Calculate duration
    local end_time=$(date +%s)
    local duration=$((end_time - start_time))
    log "INFO" "Backup completed in ${duration} seconds with status: $status"

    # Send notification
    send_summary "$backup_path" "$status"

    if [[ "$status" == "SUCCESS" ]]; then
        exit 0
    else
        exit 1
    fi
}

# Run main function with command line argument parsing
while getopts "d:j:l:h" opt; do
    case $opt in
        d) CONFIG[retention_days]="$OPTARG" ;;
        j) CONFIG[parallel_jobs]="$OPTARG" ;;
        l) CONFIG[log_file]="$OPTARG" ;;
        h)
            cat << EOF
Usage: $0 [options]

Options:
    -d DAYS    Retention days (default: ${CONFIG[retention_days]})
    -j JOBS    Parallel jobs (default: ${CONFIG[parallel_jobs]})
    -l FILE    Log file (default: ${CONFIG[log_file]})
    -h         Show this help
EOF
            exit 0
            ;;
        *) exit 1 ;;
    esac
done

# Execute main
main "$@"
```
