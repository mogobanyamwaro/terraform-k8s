Copy this into the file:

```bash
#!/bin/bash
# Simple calculator - uses multiple folders to work

echo "Simple Calculator"
echo "1. Add"
echo "2. Subtract"
read -p "Choice: " choice

read -p "First number: " num1
read -p "Second number: " num2

if [ $choice -eq 1 ]; then
    result=$((num1 + num2))
    echo "Result: $result"

    # Log the calculation (uses /var)
    echo "$(date): $num1 + $num2 = $result" >> /var/log/calc.log

    # Store in temporary storage (uses /tmp)
    echo "$result" > /tmp/last_result.txt

else
    result=$((num1 - num2))
    echo "Result: $result"
    echo "$(date): $num1 - $num2 = $result" >> /var/log/calc.log
    echo "$result" > /tmp/last_result.txt
fi

# Show last result from memory (uses /run)
if [ -f /run/last_calc.txt ]; then
    echo "Last calculation result was: $(cat /run/last_calc.txt)"
fi

# Save current result to runtime memory
echo "$result" > /run/last_calc.txt
```

## STEP 1: CREATE THE APP (Copy exactly)

# Create the app folder
mkdir -p ~/my_calculator

**Copy this ENTIRE script:**

```bash
#!/bin/bash

# ============================================
# 18-FOLDER CALCULATOR
# This app touches every Linux folder
# ============================================

# Colors for fun
GREEN='\033[0;32m'
BLUE='\033[0;34m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

echo -e "${BLUE}╔══════════════════════════════════════╗${NC}"
echo -e "${BLUE}║     18-FOLDER CALCULATOR             ║${NC}"
echo -e "${BLUE}║     Every folder tells a story       ║${NC}"
echo -e "${BLUE}╚══════════════════════════════════════╝${NC}"

# ============================================
# 1. /bin - Read essential commands
# ============================================
echo -e "\n${GREEN}[1/18] /bin - Essential Tools${NC}"
echo "Your calculator uses commands from /bin like:"
ls /bin | grep -E "^(echo|cat|ls|date|wc)" | head -5
sleep 1

# ============================================
# 2. /boot - Check system health
# ============================================
echo -e "\n${GREEN}[2/18] /boot - System Engine${NC}"
if [ -f /boot/vmlinuz* ]; then
    KERNEL=$(ls /boot/vmlinuz* | head -1 | xargs basename)
    echo "✓ Kernel exists: $KERNEL"
else
    echo "⚠ No kernel found (unusual!)"
fi
sleep 1

# ============================================
# 3. /dev - Read random number for math
# ============================================
echo -e "\n${GREEN}[3/18] /dev - Random Generator${NC}"
RANDOM_BYTE=$(head -c 1 /dev/urandom | od -An -td1 | xargs)
echo "Random number from /dev/urandom: $RANDOM_BYTE"
sleep 1

# ============================================
# 4. /etc - Read calculator settings
# ============================================
echo -e "\n${GREEN}[4/18] /etc - Settings${NC}"
CALC_CONFIG="/etc/calc_config.txt"
if [ ! -f "$CALC_CONFIG" ]; then
    echo "default_precision=2" | sudo tee "$CALC_CONFIG" > /dev/null
    echo "✓ Created config in /etc"
else
    echo "✓ Config exists in /etc"
fi
sleep 1

# ============================================
# 5. /home - Store user preferences
# ============================================
echo -e "\n${GREEN}[5/18] /home - Your Preferences${NC}"
echo "last_calc_date=$(date)" > ~/.calc_prefs
echo "✓ Saved your preferences in /home/$USER/.calc_prefs"
sleep 1

# ============================================
# 6. /lib - Show library dependencies
# ============================================
echo -e "\n${GREEN}[6/18] /lib - Shared Libraries${NC}"
echo "Bash (your calculator's shell) needs these libraries:"
ldd /bin/bash | head -3 | awk '{print "  - " $1}'
sleep 1

# ============================================
# 7. /media - Check for USB calculator backup
# ============================================
echo -e "\n${GREEN}[7/18] /media - USB Devices${NC}"
if [ -d /media/*/ 2>/dev/null ]; then
    echo "✓ USB drive detected (could backup here)"
else
    echo "ℹ No USB drives mounted (calculator runs locally)"
fi
sleep 1

# ============================================
# 8. /mnt - Temporary calculation mount
# ============================================
echo -e "\n${GREEN}[8/18] /mnt - Temporary Work${NC}"
sudo mkdir -p /mnt/calc_temp 2>/dev/null
echo "temp_result=0" | sudo tee /mnt/calc_temp/state.txt > /dev/null
echo "✓ Created temporary workspace in /mnt/calc_temp"
sleep 1

# ============================================
# 9. /opt - Optional calculator features
# ============================================
echo -e "\n${GREEN}[9/18] /opt - Advanced Features${NC}"
sudo mkdir -p /opt/calc_advanced 2>/dev/null
echo '#!/bin/bash' | sudo tee /opt/calc_advanced/sqrt.sh > /dev/null
echo 'echo "sqrt() would go here"' | sudo tee -a /opt/calc_advanced/sqrt.sh > /dev/null
sudo chmod +x /opt/calc_advanced/sqrt.sh 2>/dev/null
echo "✓ Advanced calculator features in /opt/calc_advanced"
sleep 1

# ============================================
# 10. /proc - Get live system stats
# ============================================
echo -e "\n${GREEN}[10/18] /proc - Live System Stats${NC}"
CPU_LOAD=$(cat /proc/loadavg | awk '{print $1}')
echo "Current CPU load: $CPU_LOAD (affects calculation speed)"
sleep 1

# ============================================
# 11. /root - Admin calculator mode
# ============================================
echo -e "\n${GREEN}[11/18] /root - Admin Zone${NC}"
if [ "$EUID" -eq 0 ]; then
    echo "✓ Running as ROOT - full calculator power!"
else
    echo "ℹ Running as normal user (use 'sudo' for admin features)"
fi
sleep 1

# ============================================
# 12. /run - Store current calculation session
# ============================================
echo -e "\n${GREEN}[12/18] /run - Session Memory${NC}"
echo "session_start=$(date +%s)" > /run/calc_session_$$.txt
echo "✓ Session stored in /run (will disappear on reboot)"
sleep 1

# ============================================
# 13. /sbin - System calculation tools
# ============================================
echo -e "\n${GREEN}[13/18] /sbin - System Tools${NC}"
echo "System tools available for advanced math:"
ls /sbin | grep -E "^(fdisk|mkfs|sysctl)" | head -3 | sed 's/^/  - /'
sleep 1

# ============================================
# 14. /srv - Serve calculation results
# ============================================
echo -e "\n${GREEN}[14/18] /srv - Service Output${NC}"
sudo mkdir -p /srv/calc_results 2>/dev/null
echo "Calculations served from here later" | sudo tee /srv/calc_results/readme.txt > /dev/null
echo "✓ Results can be served via web server from /srv/calc_results"
sleep 1

# ============================================
# 15. /sys - Kernel math settings
# ============================================
echo -e "\n${GREEN}[15/18] /sys - Kernel Tuning${NC}"
if [ -f /sys/devices/system/cpu/cpu0/cpufreq/scaling_cur_freq ]; then
    CPU_FREQ=$(cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_cur_freq 2>/dev/null)
    echo "✓ CPU frequency: $((CPU_FREQ/1000000)) GHz affects math speed"
else
    echo "ℹ CPU frequency info not available in VM"
fi
sleep 1

# ============================================
# 16. /tmp - Store calculation cache
# ============================================
echo -e "\n${GREEN}[16/18] /tmp - Cache Storage${NC}"
echo "0" > /tmp/calc_cache_$$.txt
echo "✓ Cache file created in /tmp (fast but temporary)"
sleep 1

# ============================================
# 17. /usr - User programs and docs
# ============================================
echo -e "\n${GREEN}[17/18] /usr - Main Program Storage${NC}"
echo "Your calculator runs using /usr/bin/bash"
echo "Documentation could live in /usr/share/doc/calc"
sleep 1

# ============================================
# 18. /var - Log all calculations
# ============================================
echo -e "\n${GREEN}[18/18] /var - Log Archive${NC}"
LOG_FILE="/var/log/calculator.log"
echo "$(date): Calculator started" | sudo tee -a "$LOG_FILE" > /dev/null
echo "✓ All calculations will be logged to $LOG_FILE"
sleep 1

# ============================================
# THE ACTUAL CALCULATOR PART
# ============================================
echo -e "\n${BLUE}════════════════════════════════════════${NC}"
echo -e "${BLUE}        LET'S CALCULATE!                ${NC}"
echo -e "${BLUE}════════════════════════════════════════${NC}"

while true; do
    echo ""
    echo "Choose operation:"
    echo "1) Add (+)"
    echo "2) Subtract (-)"
    echo "3) Multiply (*)"
    echo "4) Divide (/)"
    echo "5) Show history (from /var/log)"
    echo "6) Show session (from /run)"
    echo "7) Show cache (from /tmp)"
    echo "8) Exit"
    read -p "Choice: " choice

    if [ $choice -eq 8 ]; then
        echo "Goodbye! Your session data is in /run, logs in /var/log"
        break
    fi

    if [ $choice -eq 5 ]; then
        echo -e "\n${YELLOW}Last 5 calculations from /var/log/calculator.log:${NC}"
        sudo tail -5 /var/log/calculator.log 2>/dev/null || echo "No history yet"
        continue
    fi

    if [ $choice -eq 6 ]; then
        echo -e "\n${YELLOW}Current session (from /run):${NC}"
        cat /run/calc_session_$$.txt 2>/dev/null || echo "No session data"
        continue
    fi

    if [ $choice -eq 7 ]; then
        echo -e "\n${YELLOW}Cached values (from /tmp):${NC}"
        cat /tmp/calc_cache_$$.txt 2>/dev/null || echo "No cache"
        continue
    fi

    read -p "Enter first number: " num1
    read -p "Enter second number: " num2

    case $choice in
        1)
            result=$((num1 + num2))
            operator="+"
            ;;
        2)
            result=$((num1 - num2))
            operator="-"
            ;;
        3)
            result=$((num1 * num2))
            operator="*"
            ;;
        4)
            if [ $num2 -eq 0 ]; then
                echo "Cannot divide by zero!"
                continue
            fi
            result=$((num1 / num2))
            operator="/"
            ;;
        *)
            echo "Invalid choice"
            continue
            ;;
    esac

    echo -e "${GREEN}Result: $num1 $operator $num2 = $result${NC}"

    # Log to /var (permanent)
    echo "$(date): $num1 $operator $num2 = $result" | sudo tee -a /var/log/calculator.log > /dev/null

    # Store in /tmp (cache)
    echo "last_result=$result" > /tmp/calc_cache_$$.txt

    # Update /mnt state
    echo "last_result=$result" | sudo tee /mnt/calc_temp/state.txt > /dev/null

    # Update /run session
    echo "last_result=$result" >> /run/calc_session_$$.txt
    echo "last_operation_time=$(date +%s)" >> /run/calc_session_$$.txt

    echo ""
    echo "✅ Calculation stored in:"
    echo "   - /var/log/calculator.log (permanent)"
    echo "   - /tmp/calc_cache_$$.txt (temporary cache)"
    echo "   - /mnt/calc_temp/state.txt (temporary mount)"
    echo "   - /run/calc_session_$$.txt (session memory)"
done

echo -e "\n${BLUE}╔════════════════════════════════════════╗${NC}"
echo -e "${BLUE}║  THANK YOU FOR USING THE              ║${NC}"
echo -e "${BLUE}║  18-FOLDER CALCULATOR!                ║${NC}"
echo -e "${BLUE}║                                        ║${NC}"
echo -e "${BLUE}║  You touched ALL 18 folders!          ║${NC}"
echo -e "${BLUE}╚════════════════════════════════════════╝${NC}"

# Cleanup (optional - comment out if you want to keep data)
echo ""
read -p "Delete temporary files? (y/n): " cleanup
if [ "$cleanup" = "y" ]; then
    sudo rm -rf /mnt/calc_temp 2>/dev/null
    sudo rm -rf /opt/calc_advanced 2>/dev/null
    rm -f /run/calc_session_$$.txt 2>/dev/null
    rm -f /tmp/calc_cache_$$.txt 2>/dev/null
    echo "✓ Temporary files cleaned (logs remain in /var/log)"
fi
```
