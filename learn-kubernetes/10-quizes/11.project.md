## Complete CKA Practice Project: E-Commerce Microservices

### Project Overview

You're deploying a simple e-commerce application with:

- **Frontend** (nginx) - web UI
- **API** (redis) - session/cache storage
- **Worker** (busybox) - background jobs
- **Database** (redis) - persistent storage

This single project will test ALL 10 CKA topics in one workflow.

## Project Setup (Run these commands in order)

### Step 1: Create Namespaces (Q1 - RBAC, Q7 - Network Policies)

```bash
# Create namespaces for our application
kubectl create namespace ecom-frontend
kubectl create namespace ecom-api
kubectl create namespace ecom-database

# Verify
kubectl get namespaces | grep ecom
```

### Step 2: Create ServiceAccounts with RBAC (Q1 - RBAC, Q9 - Security)

```bash
# Create ServiceAccount for API to access database info
kubectl create serviceaccount api-sa -n ecom-api

# Create Role and RoleBinding for API ServiceAccount
cat <<EOF | kubectl apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: pod-reader
  namespace: ecom-database
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get", "list"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: api-to-database
  namespace: ecom-database
subjects:
- kind: ServiceAccount
  name: api-sa
  namespace: ecom-api
roleRef:
  kind: Role
  name: pod-reader
  apiGroup: rbac.authorization.k8s.io
EOF

# Verify RBAC
kubectl get role,rolebinding -n ecom-database
```

### Step 3: Create ConfigMap and Secret (Q3 - ConfigMap/Secret)

```bash
# Create ConfigMap for app configuration
kubectl create configmap app-config \
  --from-literal=app.properties="environment=production\nlog_level=info\ncache_ttl=300" \
  --namespace=ecom-api

# Create Secret for database password
kubectl create secret generic db-secret \
  --from-literal=password='SecurePass123!' \
  --from-literal=username='ecom_user' \
  --namespace=ecom-database

# Verify
kubectl get configmap,secret -n ecom-api
kubectl get secret -n ecom-database
```

### Step 4: Create Persistent Storage (Q6 - PV/PVC)


### Step 5: Deploy Database with SecurityContext (Q5 - Probes, Q9 - Security)

```bash
cat <<EOF | kubectl apply -f -
apiVersion: v1
kind: Pod
metadata:
  name: redis-db
  namespace: ecom-database
  labels:
    app: ecom
    tier: database
spec:
  securityContext:
    runAsNonRoot: true
    runAsUser: 999
  containers:
  - name: redis
    image: redis:alpine
    ports:
    - containerPort: 6379
    volumeMounts:
    - name: redis-storage
      mountPath: /data
    env:
    - name: REDIS_PASSWORD
      valueFrom:
        secretKeyRef:
          name: db-secret
          key: password
    livenessProbe:
      tcpSocket:
        port: 6379
      initialDelaySeconds: 15
      periodSeconds: 10
    readinessProbe:
      exec:
        command:
        - redis-cli
        - ping
      initialDelaySeconds: 5
      periodSeconds: 5
    securityContext:
      allowPrivilegeEscalation: false
      capabilities:
        drop:
        - ALL
  volumes:
  - name: redis-storage
    persistentVolumeClaim:
      claimName: redis-pvc
EOF

# Verify database is running
kubectl get pods -n ecom-database -w
```

### Step 6: Deploy API with Health Checks (Q5 - Probes)

```bash
cat <<EOF | kubectl apply -f -
apiVersion: v1
kind: Pod
metadata:
  name: api-server
  namespace: ecom-api
  labels:
    app: ecom
    tier: api
spec:
  serviceAccountName: api-sa
  containers:
  - name: redis-client
    image: redis:alpine
    command: ["/bin/sh", "-c"]
    args:
    - |
      echo "API Server Starting..."
      echo "Connected to Redis at redis-db.ecom-database.svc.cluster.local"
      while true; do
        if redis-cli -h redis-db.ecom-database.svc.cluster.local -a \$REDIS_PASSWORD ping 2>/dev/null; then
          echo "Redis is healthy - $(date)"
        else
          echo "Redis connection failed - $(date)"
        fi
        sleep 10
      done
    env:
    - name: REDIS_PASSWORD
      valueFrom:
        secretKeyRef:
          name: db-secret
          namespace: ecom-database
          key: password
    livenessProbe:
      exec:
        command:
        - sh
        - -c
        - "redis-cli -h redis-db.ecom-database.svc.cluster.local -a \$REDIS_PASSWORD ping"
      initialDelaySeconds: 20
      periodSeconds: 15
    readinessProbe:
      exec:
        command:
        - sh
        - -c
        - "redis-cli -h redis-db.ecom-database.svc.cluster.local -a \$REDIS_PASSWORD ping"
      initialDelaySeconds: 10
      periodSeconds: 5
  volumes:
  - name: config
    configMap:
      name: app-config
EOF

# Verify API pod
kubectl get pods -n ecom-api
```

### Step 7: Deploy Frontend with Deployment & Rolling Update (Q4 - Deployments)

```bash
cat <<EOF | kubectl apply -f -

  selector:

      tier: frontend
  template:
    metadata:
      labels:
        app: ecom
        tier: frontend

      containers:

            memory: "128Mi"
            cpu: "100m"

          httpGet:
            path: /
            port: 80
          initialDelaySeconds: 10
          periodSeconds: 5
        readinessProbe:
          httpGet:
            path: /
            port: 80
          initialDelaySeconds: 5
          periodSeconds: 3
EOF

# Verify deployment
kubectl get deployment,pods -n ecom-frontend
```

### Step 9: Create Network Policies (Q7 - Network Policies)

```bash
# Default deny all ingress first
cat <<EOF | kubectl apply -f -
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: default-deny-ingress
  namespace: ecom-database
spec:
  podSelector: {}
  policyTypes:
  - Ingress
---
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: allow-api-to-database
  namespace: ecom-database
spec:
  podSelector:
    matchLabels:
      tier: database
  policyTypes:
  - Ingress
  ingress:
  - from:
    - namespaceSelector:
        matchLabels:
          kubernetes.io/metadata.name: ecom-api
      podSelector:
        matchLabels:
          tier: api
    ports:
    - protocol: TCP
      port: 6379
EOF

# Verify network policies
kubectl get networkpolicy -n ecom-database
```

### Step 10: Create Background Job (Q8 - Jobs)

```bash
# Create a one-time job to initialize database
cat <<EOF | kubectl apply -f -
apiVersion: batch/v1
kind: Job
metadata:
  name: db-initializer
  namespace: ecom-database
spec:
  backoffLimit: 3
  template:
    spec:
      restartPolicy: Never
      containers:
      - name: initializer
        image: redis:alpine
        command: ["/bin/sh", "-c"]
        args:
        - |
          echo "Initializing database..."
          echo "SET app:initialized true" | redis-cli -h database-service -a \$REDIS_PASSWORD
          echo "SET app:version 1.0" | redis-cli -h database-service -a \$REDIS_PASSWORD
          echo "Database initialized at \$(date)"
        env:
        - name: REDIS_PASSWORD
          valueFrom:
            secretKeyRef:
              name: db-secret
              key: password
EOF

# Verify job completion
kubectl get jobs -n ecom-database
kubectl logs job/db-initializer -n ecom-database
```

### Step 11: Create CronJob for Backup (Q8 - CronJobs)

```bash
cat <<EOF | kubectl apply -f -
apiVersion: batch/v1
kind: CronJob
metadata:
  name: database-backup
  namespace: ecom-database
spec:
  schedule: "*/5 * * * *"  # Every 5 minutes for demo
  jobTemplate:
    spec:
      template:
        spec:
          restartPolicy: Never
          containers:
          - name: backup
            image: redis:alpine
            command: ["/bin/sh", "-c"]
            args:
            - |
              echo "Starting backup at \$(date)"
              redis-cli -h database-service -a \$REDIS_PASSWORD SAVE
              echo "Backup completed at \$(date)"
            env:
            - name: REDIS_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: db-secret
                  key: password
EOF

# Verify cronjob
kubectl get cronjob -n ecom-database
```

```bash

kind: Pod
metadata:
  name: maintenance-pod
  namespace: ecom-database
spec:
  tolerations:
  - key: "maintenance"
    operator: "Equal"
    value: "true"
    effect: "NoSchedule"
  containers:
  - name: test
    image: busybox
    command: ["sleep", "3600"]
EOF

# Remove taint after test
kubectl taint nodes <your-node> maintenance-
```
